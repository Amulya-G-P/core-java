Write an program to demonstatre a default constructor
      class Student {

        int id;
        String name;

    
    Student() 
	{
        System.out.println("Default constructor is called");
        id = 101;
        name = "Amrutha";
    }

    public  void getStudentDetails() {
        System.out.println(" The student id is ID: " + id);
        System.out.println("Name: " + name);
    }
}

class StudentRunner {
    public static void main(String[] args) {

        
        Student s1 = new Student();

    }
